    <script>
        (function () {
            const viewport = document.querySelector("[data-viewport]");
            const track = document.querySelector("[data-track]");
            const cards = Array.from(document.querySelectorAll("[data-place]"));
            const prev = document.querySelector("[data-prev]");
            const next = document.querySelector("[data-next]");
            const dotsWrap = document.querySelector("[data-dots]");

            const titleEl = document.querySelector("[data-detail-title]");
            const subEl = document.querySelector("[data-detail-sub]");
            const descEl = document.querySelector("[data-detail-desc]");
            const extraEl = document.querySelector("[data-detail-extra]");

            if (!viewport || !track || cards.length === 0) return;

            const gap = () => parseFloat(getComputedStyle(track).columnGap || getComputedStyle(track).gap || "14");
            const cardW = () => cards[0].getBoundingClientRect().width;
            const perView = () => {
                const w = viewport.clientWidth;
                if (w < 620) return 1;
                if (w < 980) return 2;
                return 3;
            };

            let index = 0;

            function pagesCount() {
                return Math.max(1, Math.ceil(cards.length / perView()));
            }

            function clamp(v, min, max) {
                return Math.max(min, Math.min(max, v));
            }

            function scrollToIndex(i) {
                index = clamp(i, 0, pagesCount() - 1);

                const step = (cardW() + gap()) * perView();
                const x = step * index;

                track.style.transform = `translateX(${-x}px)`;
                updateDots();
            }

            function setActive(card) {
                cards.forEach(c => c.classList.remove("is-active"));
                card.classList.add("is-active");

                titleEl.textContent = card.dataset.title || "";
                subEl.textContent = card.dataset.sub || "";
                descEl.textContent = card.dataset.desc || "";
                extraEl.textContent = card.dataset.extra || "";
            }

            function buildDots() {
                dotsWrap.innerHTML = "";
                const n = pagesCount();

                for (let i = 0; i < n; i++) {
                    const b = document.createElement("button");
                    b.type = "button";
                    b.className = "dot";
                    b.setAttribute("aria-label", `Go to slide ${i + 1}`);
                    b.addEventListener("click", () => scrollToIndex(i));
                    dotsWrap.appendChild(b);
                }
                updateDots();
            }

            function updateDots() {
                const dots = Array.from(dotsWrap.querySelectorAll(".dot"));
                dots.forEach((d, i) => d.classList.toggle("is-active", i === index));
            }

            prev?.addEventListener("click", () => scrollToIndex(index - 1));
            next?.addEventListener("click", () => scrollToIndex(index + 1));

            cards.forEach((card) => {
                card.addEventListener("click", () => setActive(card));
            });

            // keyboard (on the carousel area)
            viewport.addEventListener("keydown", (e) => {
                if (e.key === "ArrowLeft") { e.preventDefault(); scrollToIndex(index - 1); }
                if (e.key === "ArrowRight") { e.preventDefault(); scrollToIndex(index + 1); }
            });

            // initial
            setActive(cards[0]);
            buildDots();
            scrollToIndex(0);

            // rebuild on resize
            window.addEventListener("resize", () => {
                buildDots();
                scrollToIndex(clamp(index, 0, pagesCount() - 1));
            });
        })();
    </script>
